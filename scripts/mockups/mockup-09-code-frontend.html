Tôi là Trợ lý AI của Dragon Vision. Tôi có thể giúp bạn phân tích tồn kho, dự báo nhu cầu và đề xuất kế hoạch nhập hàng.' }]) const userInput = ref('') const isLoading = ref(false) const chatBox = ref(null) const quickStats = computed(() => [ { label: 'Tổng sản phẩm', value: store.products.length, color: 'text-blue-700' }, { label: 'Hàng sắp hết', value: store.lowStockProducts.length, color: 'text-red-600' }, { label: 'Giá trị tồn kho', value: store.totalStockValue, color: 'text-green-600' } ]) async function sendMessage() { if (!userInput.value.trim() || isLoading.value) return messages.value.push({ id: Date.now(), role: 'user', text: userInput.value }) const q = userInput.value; userInput.value = ''; isLoading.value = true await nextTick(); chatBox.value.scrollTop = chatBox.value.scrollHeight const reply = await store.askAI(q) messages.value.push({ id: Date.now(), role: 'assistant', text: reply }) isLoading.value = false await nextTick(); chatBox.value.scrollTop = chatBox.value.scrollHeight } </script>
