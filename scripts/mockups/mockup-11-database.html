-- SmartStock AI Database Schema -- Dragon Vision Technology & AI Research -- PostgreSQL 15+ CREATE TABLE product_category ( id BIGSERIAL PRIMARY KEY, code VARCHAR(20) NOT NULL UNIQUE, name VARCHAR(100) NOT NULL, description TEXT, created_at TIMESTAMP DEFAULT NOW() ); CREATE TABLE supplier ( id BIGSERIAL PRIMARY KEY, code VARCHAR(20) NOT NULL UNIQUE, name VARCHAR(150) NOT NULL, contact_name VARCHAR(100), phone VARCHAR(20), email VARCHAR(100), address TEXT ); CREATE TABLE product ( id BIGSERIAL PRIMARY KEY, sku VARCHAR(50) NOT NULL UNIQUE, name VARCHAR(200) NOT NULL, category_id BIGINT REFERENCES product_category(id), unit VARCHAR(20), purchase_price NUMERIC(15,2), selling_price NUMERIC(15,2), stock_qty INTEGER DEFAULT 0, min_stock INTEGER DEFAULT 5, status VARCHAR(20) DEFAULT 'ACTIVE' ); CREATE TABLE stock_receipt ( id BIGSERIAL PRIMARY KEY, code VARCHAR(20) NOT NULL UNIQUE, type VARCHAR(10) NOT NULL, -- IN / OUT supplier_id BIGINT REFERENCES supplier(id), total_amount NUMERIC(15,2), note TEXT, created_by VARCHAR(100), created_at TIMESTAMP DEFAULT NOW() ); CREATE TABLE stock_receipt_detail ( id BIGSERIAL PRIMARY KEY, receipt_id BIGINT NOT NULL REFERENCES stock_receipt(id), product_id BIGINT NOT NULL REFERENCES product(id), quantity INTEGER NOT NULL, unit_price NUMERIC(15,2), subtotal NUMERIC(15,2) ); CREATE TABLE inventory_history ( id BIGSERIAL PRIMARY KEY, product_id BIGINT REFERENCES product(id), change_type VARCHAR(10), -- IN / OUT qty_before INTEGER, qty_change INTEGER, qty_after INTEGER, reference_code VARCHAR(50), changed_by VARCHAR(100), changed_at TIMESTAMP DEFAULT NOW() );
